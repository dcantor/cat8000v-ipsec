# VyOS day-0 (console): identity and out-of-band management only; everything else is rendered from Nautobot
set system host-name fw-east
set system domain-name lab.local
set interfaces ethernet eth0 address 10.2.0.21/24
set interfaces ethernet eth0 description 'OOB management'
# pin every data port's name to its NIC (VyOS renames unmatched NICs at boot, which would strand the rendered configuration)
set interfaces ethernet eth1 hw-id 52:54:00:c7:09:01
set interfaces ethernet eth2 hw-id 52:54:00:c7:09:02
set interfaces ethernet eth3 hw-id 52:54:00:c7:09:03
set interfaces ethernet eth4 hw-id 52:54:00:c7:09:04
set interfaces ethernet eth5 hw-id 52:54:00:c7:09:05
set interfaces ethernet eth6 hw-id 52:54:00:c7:09:06
set interfaces ethernet eth7 hw-id 52:54:00:c7:09:07
set interfaces ethernet eth8 hw-id 52:54:00:c7:09:08
set protocols static route 10.0.0.0/8 next-hop 10.2.0.1
set service ssh port 22
set service lldp interface all
set system login user vyos authentication plaintext-password vyos
